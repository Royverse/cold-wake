`; const G_ENV = ` uniform vec3 uSun; // unit vector TOWARDS the sun uniform vec3 uSunCol; uniform vec3 uSkyUp; uniform vec3 uSkyDn; uniform vec3 uFogCol; uniform float uFogDen; uniform float uTime; uniform vec3 uCam; const G_PBR = ` float D_GGX(float NoH, float a){ float a2=a*a; float d=NoH*NoH*(a2-1.0)+1.0; return a2/(3.14159265*d*d+1e-7); } float V_Smith(float NoV, float NoL, float a){ float a2=a*a; float gv = NoL*sqrt(NoV*NoV*(1.0-a2)+a2); float gl = NoV*sqrt(NoL*NoL*(1.0-a2)+a2); return 0.5/max(gv+gl, 1e-5); } vec3 F_S(vec3 f0, float u){ return f0 + (vec3(1.0)-f0)*pow(1.0-u, 5.0); } vec3 shadePBR(vec3 N, vec3 V, vec3 albedo, float rough, float metal, float ao, float sh){ vec3 L = uSun, H = normalize(L+V); vec3 spec = F_S(f0,VoH)*D_GGX(NoH,a)*V_Smith(NoV,NoL,a); vec3 direct = (diff+spec)*uSunCol*NoL*sh; float hemi = N.y*0.5+0.5; vec3 amb = mix(uSkyDn, uSkyUp, hemi)*albedo*(1.0-metal)*ao; vec3 R = reflect(-V,N); vec3 ambS = skyColor(R)*F_S(f0,NoV)*(1.0-rough*0.8)*ao; return direct + amb*0.60 + ambS*0.42; } `; const G_SHADOW = ` uniform sampler2D uShadow; uniform mat4 uShadowMat; uniform float uShadowTexel; uniform float uShadowOn; float shadowAt(vec3 wp, float NoL){ if(uShadowOn < 0.5) return 1.0; vec4 sp = uShadowMat*vec4(wp,1.0); vec3 p = sp.xyz/sp.w*0.5+0.5; if(p.x<0.004||p.x>0.996||p.y<0.004||p.y>0.996||p.z>1.0) return 1.0; } return s/9.0; } `; /* ---------------------------- sky ---------------------------------- */ const VS_FS = G_HEAD + ` out vec2 vUv; void main(){ vec2 p = vec2(float((gl_VertexID<<1)&2), float(gl_VertexID&2)); /* cheap layered cloud slabs: 3 parallax planes with a 2-tap light term */ float cloudLayer(vec3 rd, float h, float scale, float t, out float lit){ lit = 0.0; if(rd.y < 0.015) return 0.0; float d = h/rd.y; vec2 p = (uCam.xz + rd.xz*d)*scale + vec2(t*0.004, t*0.0016); float n = fbm2(p); float n2_ = fbm2(p*2.7 + vec2(n*0.6)); float dens = smoothstep(0.52 - uCloudCover*0.22, 0.86, n*0.72 + n2_*0.28); // light march: sample once towards the sun for a silver lining vec2 ls = p + uSun.xz*0.06; float nl = fbm2(ls)*0.72 + fbm2(ls*2.7)*0.28; lit = clamp(1.0 - smoothstep(0.45, 0.85, nl), 0.0, 1.0); float fade = smoothstep(0.015, 0.10, rd.y); return dens*fade; } void main(){ vec4 hp = uInvVP*vec4(vUv*2.0-1.0, 1.0, 1.0); vec3 rd = normalize(hp.xyz/hp.w - uCam); vec3 col = skyColor(rd); // sun disc with soft limb float mu = dot(rd, uSun); col += uSunCol*smoothstep(0.9993, 0.99975, mu)*7.0; /* distant massif silhouettes (three ridgelines at infinity) */ float az = atan(rd.z, rd.x); for(int i=0;i<3;i++){ float fi = float(i); float freq = 1.6 + fi*2.1; float rl = fbm2(vec2(az*freq*0.16 + fi*7.3, 0.31))*0.9 + fbm2(vec2(az*freq*0.5 + fi*3.1, 0.77))*0.35; float top = 0.012 + rl*(0.115 - fi*0.028); float hx = fbm2(uvA + e.xy*0.055)*0.55 + n2(uvB + e.xy*0.09)*0.45; float hz = fbm2(uvA + e.yx*0.055)*0.55 + n2(uvB + e.yx*0.09)*0.45; vec3 bump = vec3(h0-hx, 0.85, h0-hz); float bumpAmt = mix(1.4, 0.35, vI.x); N = normalize(N + vec3(bump.x, 0.0, bump.z)*bumpAmt*clamp(1.0-dist*0.004, 0.15, 1.0)); /* --- material blend: dark wet rock -> ice -> snow --- */ float up = clamp(N.y, 0.0, 1.0); float snowW = clamp(vI.x*0.75 + smoothstep(0.55, 0.88, up)*0.75 - dtA*0.35, 0.0, 1.0); snowW = smoothstep(0.22, 0.62, snowW); vec3 rock = mix(vec3(0.030,0.035,0.042), vec3(0.075,0.084,0.096), dtA); rock = mix(rock, vec3(0.055,0.085,0.115), smoothstep(0.4,0.9,dtB)*0.5); // wet basalt vec3 ice = mix(vec3(0.24,0.36,0.45), vec3(0.46,0.60,0.70), dtB); vec3 snow = mix(vec3(0.56,0.63,0.74), vec3(0.76,0.83,0.92), grain); float iceW = smoothstep(0.30, 0.75, vI.x)*(1.0-snowW)*0.85; vec3 albedo = mix(rock, ice, iceW); albedo = mix(albedo, snow, snowW); float rough = mix(mix(0.82, 0.30, iceW), 0.62, snowW); }`; /* ----------------------------- props -------------------------------- */ /* Shared by every rigid mesh: hulls, riders, drums, ice blocks, rotors. */ const VS_PROP = G_HEAD + ` layout(location=0) in vec3 aPos;
