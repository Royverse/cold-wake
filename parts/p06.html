/* ============================ 04 track ============================= */ /* The canyon is a 5 m sampled spline. Everything in the world - terrain, AI steering, obstacle placement, mission beats - is addressed by (distance along track `s`, signed lateral offset `lat`). */ const STEP = 5; // metres between track samples const TRACK_LEN = 20800; // ~20.8 km run const BED = 7.5; // riverbed depth below sea level /** zones where the canyon closes into an ice tunnel [startS, endS] */ const CAVE_ZONES = [[8450, 8980], [15250, 15620]]; /** hard narrows: the corridor pinches to a slalom gate */ const GATE_S = [1750, 3100, 4650, 6150, 7400, 9700, 11200, 12750, 14100, 16400, 17900, 19200]; const Track = { n: (TRACK_LEN / STEP) | 0, x: null, z: null, dx: null, dz: null, w: null, wl: null, wr: null, cave: null, hd: null, build() { const n = this.n; this.x = new Float32Array(n); this.z = new Float32Array(n); this.dx = new Float32Array(n); this.dz = new Float32Array(n); this.w = new Float32Array(n); this.wl = new Float32Array(n); this.wr = new Float32Array(n); this.cave = new Float32Array(n); this.hd = new Float32Array(n); let px = 0, pz = 0, h = 0; for (let i = 0; i < n; i++) { const s = i * STEP; // curvature: lazy sweeps + occasional hard corners late in the run const prog = s / TRACK_LEN; let k = (fbm(s * 0.0016, 11.7, 3) - .5) * 0.0072; k += (fbm(s * 0.006, 77.3, 2) - .5) * 0.0042 * (0.35 + prog); const tight = smoothstep(.55, .95, fbm(s * 0.0009, 133.1, 2)); k *= 1 + tight * 1.9; if (s < 260) k *= 0.15; // calm start for the intro h += k * STEP; this.hd[i] = h; this.dx[i] = sin(h); this.dz[i] = cos(h); this.x[i] = px; this.z[i] = pz; px += this.dx[i] * STEP; pz += this.dz[i] * STEP; // corridor half width let w = 30 + (fbm(s * 0.0021, 5.5, 3) - .5) * 22; w *= 1 - prog * 0.18; // squeeze as the run escalates for (let g = 0; g < GATE_S.length; g++) { const d = abs(s - GATE_S[g]); if (d < 240) w *= lerp(1, 0.42, smoothstep(240, 40, d)); } this.w[i] = clamp(w, 9, 46); // wall heights (independent per side) this.wl[i] = 26 + fbm(s * 0.0013, 301.7, 3) * 74; this.wr[i] = 26 + fbm(s * 0.0013, 811.2, 3) * 74; let cv = 0; for (const z of CAVE_ZONES) { if (s > z[0] - 90 && s < z[1] + 90) cv = max(cv, smoothstep(z[0] - 90, z[0] + 60, s) * (1 - smoothstep(z[1] - 60, z[1] + 90, s))); } this.cave[i] = cv; } }, idx(s) { return clamp((s / STEP) | 0, 0, this.n - 1); }, /** world position of the centreline at distance s */ posAt(s, o) { const f = clamp(s / STEP, 0, this.n - 1.001), i = f | 0, t = f - i; o = o || [0, 0, 0]; o[0] = lerp(this.x[i], this.x[i + 1], t); o[1] = 0; o[2] = lerp(this.z[i], this.z[i + 1], t); return o; }, dirAt(s, o) { const i = this.idx(s); o = o || [0, 0]; o[0] = this.dx[i]; o[1] = this.dz[i]; return o; }, headAt(s) { return this.hd[this.idx(s)]; }, widthAt(s) { const f = clamp(s / STEP, 0, this.n - 1.001), i = f | 0, t = f - i; return lerp(this.w[i], this.w[i + 1], t); }, caveAt(s) { return this.cave[this.idx(s)]; }, /** project a world xz onto the track. `hint` is the last known index. */ project(x, z, hint, out) { out = out || { s: 0, lat: 0, i: 0 }; const n = this.n; let bi = clamp(hint | 0, 0, n - 2), bd = 1e18, bt = 0; const lo = max(0, bi - 26), hi = min(n - 2, bi + 60); for (let i = lo; i <= hi; i++) { const ax = this.x[i], az = this.z[i]; const bx = this.x[i + 1] - ax, bz = this.z[i + 1] - az; let t = ((x - ax) * bx + (z - az) * bz) / (bx * bx + bz * bz); t = clamp(t, 0, 1); const px = ax + bx * t - x, pz = az + bz * t - z; const d = px * px + pz * pz; if (d < bd) { bd = d; bi = i; bt = t; } } const ax = this.x[bi], az = this.z[bi], dx = this.dx[bi], dz = this.dz[bi]; out.i = bi; out.s = (bi + bt) * STEP; out.lat = (x - ax) * dz - (z - az) * dx; // right = (dz,-dx) return out; }, /** world position from track coordinates */ toWorld(s, lat, o) { const i = this.idx(s), f = clamp(s / STEP, 0, this.n - 1.001), fi = f | 0, t = f - fi; const cx = lerp(this.x[fi], this.x[fi + 1], t), cz = lerp(this.z[fi], this.z[fi + 1], t); o = o || [0, 0, 0]; o[0] = cx + this.dz[i] * lat; o[1] = 0; o[2] = cz - this.dx[i] * lat; return o; }, }; /* ============================ 05 terrain =========================== */ /* Canyon shells are streamed in 200 m chunks around the player. Vertex layout: pos(3) nrm(3) info(2 = ice/snow exposure, cavity AO). */ const PROFILE = (() => { const half = [0, .11, .26, .42, .58, .73, .84, .92, .98, 1.04, 1.11, 1.19, 1.28, 1.39, 1.52, 1.68, 1.9, 2.18, 2.55, 3.1, 4.4]; const a = []; for (let i = half.length - 1; i > 0; i--) a.push(-half[i]); for (let i = 0; i < half.length; i++) a.push(half[i]); return a; // 41 lateral samples })(); const CHUNK_S = 40; // samples per chunk (200 m) const CHUNK_LEN = CHUNK_S * STEP; /** height of the canyon shell at |u| (before noise), in metres */ function shellY(au, W, wallH) { if (au < .94) return -BED * (1 - au * au * .72); if (au < 1.16) { const t = smoothstep(0, 1, (au - .94) / .22); return lerp(-BED * (1 - .94 * .94 * .72), 1.5, t); } const t = clamp((au - 1.16) / 1.05, 0, 1); let y = 1.5 + wallH * (1 - Math.pow(1 - t, 2.35)); if (au > 2.21) y += (au - 2.21) * W * 0.34; return y; } const _tv = [0, 0, 0]; /** world position + material weights for terrain vertex (i, j) */ function terrainVert(i, j, out) { i = clamp(i, 0, Track.n - 1); const jc = clamp(j, 0, PROFILE.length - 1); const u = PROFILE[jc]; const au = abs(u); const W = Track.w[i], wallH = u < 0 ? Track.wl[i] : Track.wr[i]; const cx = Track.x[i], cz = Track.z[i], dx = Track.dx[i], dz = Track.dz[i]; let lat = u * W; let y = shellY(au, W, wallH); // wall blend factor: 0 in the channel, 1 up the cliff const t = clamp((au - 1.0) / 0.9, 0, 1); const wallT = t * t * (3 - 2 * t); // ice tunnel: fold the outer profile up and over the channel const cave = Track.cave[i]; if (cave > 0.001 && au > 1.04) { const q = clamp((au - 1.04) / (4.4 - 1.04), 0, 1); const th = q * PI * 0.5; const lipY = shellY(1.04, W, wallH); const archH = 16 + W * 0.5 + wallH * 0.06; const aLat = sign(u) * W * 1.04 * cos(th * 1.02); const aY = lipY + archH * sin(th) - q * q * 2.0; lat = lerp(lat, aLat, cave); y = lerp(y, aY, cave); } // world-space erosion noise (applied on the shell only) let px = cx + dz * lat, pz = cz - dx * lat; const n1 = fbm(px * .028, pz * .028, 3) - .5; const n2 = vnoise(px * .13, pz * .13) - .5; const n3 = vnoise(px * .5, pz * .5) - .5; const amp = wallT * (1 - cave * .55); y += n1 * wallH * .30 * amp + n2 * 5.5 * amp + n3 * 1.1 * amp; const latPush = (n1 * W * .40 + n2 * 3.2) * amp; lat += latPush; px = cx + dz * lat; pz = cz - dx * lat; // gentle drift dunes on the flat snow shelf if (au > .9 && au < 1.3) y += (vnoise(px * .09, pz * .09) - .5) * 1.4; out = out || _tv; out[0] = px; out[1] = y; out[2] = pz; return out; } /** ice/snow exposure + cavity AO for a vertex */ function terrainInfo(i, j, y, ny) { const u = PROFILE[clamp(j, 0, PROFILE.length - 1)], au = abs(u); const wallH = u < 0 ? Track.wl[i] : Track.wr[i]; let ice; if (au < .9) ice = .05 + smoothstep(-BED, 0.5, y) * .35; else if (au < 1.2) ice = .92; else ice = clamp(.42 + .5 * clamp(y / max(wallH, 1), 0, 1), 0, 1); ice = clamp(ice + (vnoise(i * .21 + u * 7.7, u * 3.3) - .5) * .3 + ny * .18, 0, 1); const open = clamp((y + BED) / (wallH * .72 + 8), 0, 1); let ao = clamp(.32 + .68 * Math.pow(open, .7), .25, 1); ao *= lerp(1, .78, Track.cave[i]); return [ice, ao]; } const Terrain = { chunks: new Map(), pending: [], cols: PROFILE.length, buildChunk(ci) { const rows = CHUNK_S + 1, cols = this.cols; const i0 = ci * CHUNK_S; if (i0 >= Track.n - 1) return null; // past the end of the spline const verts = new Float32Array(rows * cols * 8); const a = [0, 0, 0], b = [0, 0, 0], c = [0, 0, 0], d = [0, 0, 0], p = [0, 0, 0]; let mnx = 1e9, mny = 1e9, mnz = 1e9, mxx = -1e9, mxy = -1e9, mxz = -1e9; let o = 0; for (let r = 0; r < rows; r++) { const i = i0 + r; if (i >= Track.n) break; for (let j = 0; j < cols; j++) { terrainVert(i, j, p); terrainVert(i, j - 1, a); terrainVert(i, j + 1, b); terrainVert(i - 1, j, c); terrainVert(i + 1, j, d); const ux = b[0] - a[0], uy = b[1] - a[1], uz = b[2] - a[2]; const vx = d[0] - c[0], vy = d[1] - c[1], vz = d[2] - c[2]; let nx = vy * uz - vz * uy, ny = vz * ux - vx * uz, nz = vx * uy - vy * ux; const il = 1 / (Math.hypot(nx, ny, nz) || 1); nx *= il; ny *= il; nz *= il; const inf = terrainInfo(i, j, p[1], ny); verts[o++] = p[0]; verts[o++] = p[1]; verts[o++] = p[2]; verts[o++] = nx; verts[o++] = ny; verts[o++] = nz; verts[o++] = inf[0]; verts[o++] = inf[1]; if (p[0] < mnx) mnx = p[0]; if (p[0] > mxx) mxx = p[0]; if (p[1] < mny) mny = p[1]; if (p[1] > mxy) mxy = p[1]; if (p[2] < mnz) mnz = p[2]; if (p[2] > mxz) mxz = p[2]; } } const nRows = min(rows, Track.n - i0); const idx = new Uint32Array((nRows - 1) * (cols - 1) * 6); let k = 0; for (let r = 0; r < nRows - 1; r++) for (let j = 0; j < cols - 1; j++) { const v0 = r * cols + j, v1 = v0 + 1, v2 = v0 + cols, v3 = v2 + 1; idx[k++] = v0; idx[k++] = v2; idx[k++] = v1; idx[k++] = v1; idx[k++] = v2; idx[k++] = v3; } const m = mkMeshGL(verts, idx, [{ n: 3 }, { n: 3 }, { n: 2 }]); m.center = [(mnx + mxx) * .5, (mny + mxy) * .5, (mnz + mxz) * .5]; m.radius = Math.hypot(mxx - mnx, mxy - mny, mxz - mnz) * .5 + 2; m.ci = ci; return m; }, /** keep chunks around the player; build at most `budget` per frame */ update(s, budget) { const last = floor((Track.n - 2) / CHUNK_S); const c0 = max(0, floor((s - 700) / CHUNK_LEN)); const c1 = min(last, floor((s + 2100) / CHUNK_LEN)); for (const [ci, m] of this.chunks) { if (ci < c0 - 1 || ci > c1 + 1) { delMeshGL(m); this.chunks.delete(ci); } } let made = 0; for (let ci = c0; ci <= c1 && made < budget; ci++) { if (this.chunks.has(ci)) continue; const m = this.buildChunk(ci); if (m) this.chunks.set(ci, m); made++; } return made; }, draw(frustum) { for (const m of this.chunks.values()) { if (frustum && !frustum.sphere(m.center, m.radius)) continue; gl.bindVertexArray(m.vao); gl.drawElements(gl.TRIANGLES, m.count, gl.UNSIGNED_INT, 0); } }, clear() { for (const m of this.chunks.values()) delMeshGL(m); this.chunks.clear(); }, }; /* ---- frustum culling ---- */ const Frustum = { p: new Float32Array(24), fromVP(m) { const p = this.p; for (let i = 0; i < 6; i++) { const s = (i % 2) ? -1 : 1, r = (i / 2) | 0; p[i * 4] = m[3] + s * m[r]; p[i * 4 + 1] = m[7] + s * m[4 + r]; p[i * 4 + 2] = m[11] + s * m[8 + r]; p[i * 4 + 3] = m[15] + s * m[12 + r]; const l = Math.hypot(p[i * 4], p[i * 4 + 1], p[i * 4 + 2]) || 1; p[i * 4] /= l; p[i * 4 + 1] /= l; p[i * 4 + 2] /= l; p[i * 4 + 3] /= l; } return this; }, sphere(c, r) { const p = this.p; for (let i = 0; i < 6; i++) { if (p[i * 4] * c[0] + p[i * 4 + 1] * c[1] + p[i * 4 + 2] * c[2] + p[i * 4 + 3] < -r) return false; } return true; }, };
